Create Custom Workflows

Design your own workflow patterns using Mao's declarative configuration system:

workflows/custom_analysis.json
{
  "name": "custom_analysis",
  "description": "Custom analysis workflow",
  "steps": [
    {
      "name": "initial_research",
      "agent": "research_specialist",
      "tools": ["web_search", "document_parser"]
    },
    {
      "name": "parallel_analysis",
      "type": "parallel",
      "agents": ["data_analyst", "market_researcher", "trend_analyst"],
      "depends_on": ["initial_research"]
    },
    {
      "name": "synthesis",
      "agent": "synthesis_specialist",
      "tools": ["report_generator"],
      "inputs": ["initial_research", "parallel_analysis"]
    }
  ],
  "memory_enabled": true,
  "cost_limit": 2.00
}
