class QualityAssuranceValidator:
    def __init__(self):
        self.memory_mcp = MemoryMCPManager()
        
    def validate_phase_deliverables(self, phase_name: str, deliverables: list, success_criteria: list):
        """Validate phase deliverables against success criteria"""
        
        validation_results = {
            "phase": phase_name,
            "timestamp": datetime.now().isoformat(),
            "criteria_met": 0,
            "total_criteria": len(success_criteria),
            "quality_score": 0.0,
            "insights": [],
            "recommendations": []
        }
        
        # Validate each criterion
        for criterion in success_criteria:
            is_met = self._evaluate_criterion(criterion, deliverables)
            if is_met:
                validation_results["criteria_met"] += 1
                validation_results["insights"].append(f"✅ {criterion} - Successfully achieved")
            else:
                validation_results["recommendations"].append(f"❌ {criterion} - Needs improvement")
        
        # Calculate quality score
        if validation_results["total_criteria"] > 0:
            validation_results["quality_score"] = (validation_results["criteria_met"] / validation_results["total_criteria"]) * 10
        
        return validation_results
    
    def _evaluate_criterion(self, criterion: str, deliverables: list):
        """Evaluate if a specific criterion is met by the deliverables"""
        
        # Simple heuristic evaluation (could be enhanced with AI analysis)
        criterion_lower = criterion.lower()
        
        # Check for keyword matches in deliverable content
        for deliverable in deliverables:
            content = deliverable.get("content", "").lower()
            name = deliverable.get("name", "").lower()
            
            # Look for criterion keywords in content or filename
            if any(keyword in content or keyword in name for keyword in criterion_lower.split()):
                return True
        
        return False

class CommandRegistry:
    def __init__(self):
        self.memory_mcp = MemoryMCPManager()
        self.registry_file = "configs/cli/custom_commands.json"
        
    def register_command(self, command_name: str, command_entry: dict):
        """Register custom command with workflow ID tracking"""
        registry = self._load_registry()
        registry[command_name] = command_entry
        self._save_registry(registry)
        
        # Track in Memory MCP
        self.memory_mcp.create_entities([{
            "name": f"custom-command-{command_name}",
            "entityType": "cli-command",
            "observations": [
                f"Command: mao {command_name}",
                f"Workflow ID: {command_entry['workflow_id']}",
                f"Created: {command_entry['created']}"
            ]
        }])
    
    def get_command(self, command_name: str):
        """Retrieve command entry by name"""
        registry = self._load_registry()
        return registry.get(command_name)
    
    def list_commands(self):
        """List all registered custom commands"""
        registry = self._load_registry()
        return list(registry.values())
    
    def _load_registry(self):
        """Load command registry from file"""
        try:
            with open(self.registry_file, 'r') as f:
                return json.load(f)
        except FileNotFoundError:
            return {}
    
    def _save_registry(self, registry: dict):
        """Save command registry to file"""
        os.makedirs(os.path.dirname(self.registry_file), exist_ok=True)
        with open(self.registry_file, 'w') as f:
            json.dump(registry, f, indent=2)

class WorkflowStateManager:
    def __init__(self):
        self.memory_mcp = MemoryMCPManager()
        self.files_api = FilesAPIManager()
        
    def create_workflow_state(self, workflow_id: str, initial_config: dict):
        """Create comprehensive workflow state"""
        state = {
            "workflow_id": workflow_id,
            "created": datetime.now().isoformat(),
            "status": "initialized",
            "current_phase": 0,
            "total_phases": len(initial_config.get("phases", [])),
            "progress_percentage": 0,
            "config": initial_config,
            "execution_log": [],
            "deliverables": [],
            "cost_tracking": {
                "estimated": 0.0,
                "actual": 0.0,
                "breakdown": {}
            }
        }
        
        # Save initial state
        self._save_workflow_state(workflow_id, state)
        
        return state
    
    def update_workflow_progress(self, workflow_id: str, phase_completed: str, results: dict):
        """Update workflow progress and state"""
        state = self._load_workflow_state(workflow_id)
        
        # Update progress
        state["current_phase"] += 1
        state["progress_percentage"] = (state["current_phase"] / state["total_phases"]) * 100
        
        # Add to execution log
        state["execution_log"].append({
            "timestamp": datetime.now().isoformat(),
            "phase": phase_completed,
            "status": results.get("status", "completed"),
            "deliverables": results.get("deliverables", []),
            "cost": results.get("cost", 0.0)
        })
        
        # Update cost tracking
        if results.get("cost"):
            state["cost_tracking"]["actual"] += results["cost"]
            state["cost_tracking"]["breakdown"][phase_completed] = results["cost"]
        
        # Add deliverables
        if results.get("deliverables"):
            state["deliverables"].extend(results["deliverables"])
        
        # Update status
        if state["current_phase"] >= state["total_phases"]:
            state["status"] = "completed"
        else:
            state["status"] = "in_progress"
        
        # Save updated state
        self._save_workflow_state(workflow_id, state)
        
        # Update Memory MCP
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Progress: {state['progress_percentage']:.1f}% - Phase {phase_completed} completed"
        )
        
        return state
    
    def recover_workflow_session(self, workflow_id: str):
        """Recover workflow state after interruption"""
        
        # Get state from Memory MCP
        workflow_context = self.memory_mcp.get_workflow_context(workflow_id)
        if not workflow_context:
            return None
        
        # Load detailed state
        state = self._load_workflow_state(workflow_id)
        
        # Determine recovery point
        recovery_info = {
            "workflow_id": workflow_id,
            "last_completed_phase": state.get("current_phase", 0) - 1,
            "next_phase": state.get("current_phase", 0),
            "progress": state.get("progress_percentage", 0),
            "deliverables_so_far": state.get("deliverables", []),
            "can_resume": state.get("status") in ["in_progress", "paused"],
            "recovery_notes": self._generate_recovery_notes(state)
        }
        
        # Track recovery attempt
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Session recovery attempted - {recovery_info['can_resume']}"
        )
        
        return recovery_info
    
    def _save_workflow_state(self, workflow_id: str, state: dict):
        """Save workflow state to Files API"""
        state_json = json.dumps(state, indent=2)
        
        # Save to Files API (via Code Execution for accessibility)
        file_id = self.files_api.save_workflow_file(
            workflow_id=workflow_id,
            filename="workflow_state.json",
            content=state_json
        )
        
        return file_id
    
    def _load_workflow_state(self, workflow_id: str):
        """Load workflow state from Files API"""
        try:
            state_content = self.files_api.load_workflow_file(
                workflow_id=workflow_id,
                filename="workflow_state.json"
            )
            return json.loads(state_content)
        except FileNotFoundError:
            # Create default state if not found
            return {
                "workflow_id": workflow_id,
                "status": "unknown",
                "current_phase": 0,
                "progress_percentage": 0,
                "execution_log": [],
                "deliverables": []
