```python
# orchestrator/memory_mcp.py
class MemoryMCPManager:
    def __init__(self):
        self.client = MCPClient("memory")
        
    def create_workflow_context(self, workflow_id: str, user_goal: str):
        """Create workflow entity with unique ID"""
        entity_data = {
            "name": f"workflow-{workflow_id}",
            "entityType": "active-workflow", 
            "observations": [
                f"User goal: {user_goal}",
                f"Created: {datetime.now().isoformat()}",
                "Status: initialized"
            ]
        }
        return self.client.create_entities([entity_data])
    
    def update_workflow_state(self, workflow_id: str, state_update: str):
        """Add observations to workflow entity"""
        return self.client.add_observations([{
            "entityName": f"workflow-{workflow_id}",
            "contents": [state_update]
        }])
    
    def get_workflow_context(self, workflow_id: str):
        """Retrieve full workflow context"""
        return self.client.open_nodes([f"workflow-{workflow_id}"])
    
    def search_workflow_patterns(self, query: str):
        """Find similar workflows for pattern matching"""
        return self.client.search_nodes(query)
        
    def handle_session_recovery(self, workflow_id: str):
        """Restore workflow state after interruption"""
        context = self.get_workflow_context(workflow_id)
        if context:
            return self._parse_workflow_state(context)
        return None
```

### **Integration Points**
- `orchestrator/core.py` imports and initializes MemoryMCPManager
- All workflow state changes call `update_workflow_state()`
- Agent handoffs include workflow context via `get_workflow_context()`
- Session recovery uses `handle_session_recovery()`

```python
# orchestrator/files_api.py
class FilesAPIManager:
    def __init__(self):
        self.client = AnthropicFiles()
        self.memory_mcp = MemoryMCPManager()
    
    def create_workflow_workspace(self, workflow_id: str):
        """Create file structure for workflow"""
        workspace_structure = {
            "config": f"workflow-{workflow_id}-config.json",
            "drafts": f"workflow-{workflow_id}-drafts/",
            "handoffs": f"workflow-{workflow_id}-handoffs/",
            "final": f"workflow-{workflow_id}-deliverables/"
        }
        
        # Store workspace info in Memory MCP
        self.memory_mcp.update_workflow_state(
            workflow_id, 
            f"Workspace created: {workspace_structure}"
        )
        return workspace_structure
    
    def save_draft(self, workflow_id: str, content: str, draft_type: str):
        """Save draft with automatic versioning"""
        version = self._get_next_version(workflow_id, draft_type)
        filename = f"workflow-{workflow_id}-{draft_type}-v{version}"
        file_id = self.client.upload(content, filename)
        
        # Update Memory MCP with draft info
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Draft saved: {draft_type} v{version} -> {file_id}"
        )
        return file_id
    
    def prepare_agent_handoff(self, workflow_id: str, agent_materials: dict):
        """Package materials for agent handoff"""
        handoff_package = {
            "workflow_id": workflow_id,
            "context": self.memory_mcp.get_workflow_context(workflow_id),
            "materials": agent_materials,
            "return_instructions": f"Call back with workflow-{workflow_id}",
            "handoff_timestamp": datetime.now().isoformat()
        }
        
        file_id = self.client.upload(
            json.dumps(handoff_package, indent=2), 
            f"handoff-{workflow_id}-{uuid4().hex[:8]}"
        )
        
        # Track handoff in Memory MCP
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Agent handoff prepared: {file_id}"
        )
        return file_id
    
    def restore_agent_context(self, workflow_id: str, handoff_id: str):
        """Restore context when agent returns"""
        handoff_data = self.client.download(handoff_id)
        package = json.loads(handoff_data)
        
        # Update workflow state with return
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Agent returned from handoff: {handoff_id}"
        )
        return package
```

### **Integration Points**
- Workflow creation automatically sets up file workspace
- All drafts saved with workflow context tracking
- Agent handoffs include complete context packages
- Memory MCP tracks all file operations for recovery

```python
# orchestrator/mcp_connector.py
class MCPConnector:
    def __init__(self):
        self.servers = {}
        self.memory_manager = MemoryMCPManager()
        self.registered_tools = {}
        
    def register_server(self, server_config: dict):
        """Register external MCP server"""
        server_name = server_config["name"]
        self.servers[server_name] = MCPServerConnection(server_config)
        
        # Discover available tools
        tools = self.servers[server_name].list_tools()
        self.registered_tools[server_name] = tools
        
        # Log server registration in Memory MCP
        self.memory_manager.create_entities([{
            "name": f"mcp-server-{server_name}",
            "entityType": "mcp-server",
            "observations": [
                f"Registered: {server_config}",
                f"Available tools: {list(tools.keys())}"
            ]
        }])
        
        return tools
    
    def execute_tool(self, server_name: str, tool_name: str, params: dict, workflow_id: str = None):
        """Execute tool on external MCP server"""
        if server_name not in self.servers:
            raise ValueError(f"Server {server_name} not registered")
            
        if tool_name not in self.registered_tools[server_name]:
            raise ValueError(f"Tool {tool_name} not available on {server_name}")
            
        result = self.servers[server_name].execute_tool(tool_name, params)
        
        # Log tool execution if part of workflow
        if workflow_id:
            self.memory_manager.update_workflow_state(
                workflow_id,
                f"MCP tool executed: {server_name}.{tool_name} -> Success"
            )
            
        return result
    
    def get_available_tools(self):
        """Get all available tools across all servers"""
        all_tools = {}
        for server_name, tools in self.registered_tools.items():
            for tool_name, tool_info in tools.items():
                all_tools[f"{server_name}.{tool_name}"] = tool_info
        return all_tools
```

### **Integration Points**
- `orchestrator/core.py` initializes and registers default servers
- `orchestrator/manager_tools.py` uses connector for tool discovery
- Tool execution automatically logged in workflow context
- Dynamic tool availability updates tool manager
