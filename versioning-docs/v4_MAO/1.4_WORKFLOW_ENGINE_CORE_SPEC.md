# orchestrator/setup_bridge.py
class SetupScriptBridge:
    def __init__(self):
        self.memory_mcp = MemoryMCPManager()
        self.files_api = FilesAPIManager()
        self.script_generator = SetupScriptGenerator()
        
    def create_workflow_from_conversation(self, user_goal: str, conversation_context: dict):
        """Transform conversation into executable workflow"""
        workflow_id = f"workflow-{uuid4().hex[:8]}"
        
        # Create workflow entity in Memory MCP
        self.memory_mcp.create_workflow_context(workflow_id, user_goal)
        
        # Analyze conversation to extract requirements
        workflow_spec = self._analyze_conversation(user_goal, conversation_context)
        
        # Generate use-case JSON config
        use_case_config = self._generate_use_case_config(workflow_id, workflow_spec)
        
        # Create setup script
        setup_script = self.script_generator.create_setup_script(
            workflow_id=workflow_id,
            config=use_case_config,
            user_goal=user_goal
        )
        
        # Generate custom command (spaces not hyphens!)
        custom_command = self._generate_custom_command(workflow_spec)
        
        # Create workspace structure
        workspace = self._create_workflow_workspace(workflow_id, workflow_spec)
        
        # Generate README
        readme_content = self._generate_use_case_readme(
            workflow_id=workflow_id,
            spec=workflow_spec,
            command=custom_command
        )
        
        # Save everything to Files API
        file_ids = {
            "config": self.files_api.save_workflow_file(
                workflow_id, "config.json", json.dumps(use_case_config, indent=2)
            ),
            "setup_script": self.files_api.save_workflow_file(
                workflow_id, "setup.sh", setup_script
            ),
            "readme": self.files_api.save_workflow_file(
                workflow_id, "README.md", readme_content
            )
        }
        
        # Track creation in Memory MCP
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Workflow created from conversation: {custom_command}"
        )
        
        return {
            "workflow_id": workflow_id,
            "custom_command": custom_command,
            "workspace": workspace,
            "file_ids": file_ids,
            "ready_to_execute": True
        }
    
    def _analyze_conversation(self, user_goal: str, context: dict):
        """Extract workflow requirements from conversation"""
        return {
            "goal": user_goal,
            "complexity": self._estimate_complexity(user_goal),
            "required_tools": self._identify_required_tools(user_goal),
            "estimated_phases": self._plan_phases(user_goal),
            "success_criteria": self._define_success_criteria(user_goal),
            "constraints": context.get("constraints", {}),
            "preferences": context.get("preferences", {})
        }
    
    def _generate_custom_command(self, workflow_spec: dict):
        """Generate human-readable command with spaces"""
        goal_words = workflow_spec["goal"].lower().split()
        
        # Clean and format words (spaces not hyphens!)
        command_words = []
        for word in goal_words[:4]:  # Limit to 4 words for readability
            clean_word = re.sub(r'[^a-z0-9]', '', word)
            if len(clean_word) > 2:  # Skip very short words
                command_words.append(clean_word)
        
        return " ".join(command_words)
    
    def _generate_use_case_readme(self, workflow_id: str, spec: dict, command: str):
        """Generate comprehensive README for use case"""
        return f"""# {spec['goal']}
**Workflow ID:** {workflow_id}
**Command:** `mao {command}`

## Overview
{spec['goal']}

## Estimated Execution
- **Phases:** {len(spec['estimated_phases'])}
- **Tools Required:** {', '.join(spec['required_tools'])}
- **Complexity:** {spec['complexity']}

## Success Criteria
{chr(10).join(f"- {criteria}" for criteria in spec['success_criteria'])}

## Quick Actions
```bash
# Execute workflow
mao {command}

# Check status
mao status {workflow_id}

# Resume if interrupted
mao resume {workflow_id}

# View results
mao results {workflow_id}
```

## Generated Files
This workflow will create organized deliverables in phase-based structure.

---
*Generated by MAO v4 - Modular Agent Orchestrator*
"""
```

# orchestrator/command_engine.py
class CommandEngine:
    def __init__(self):
        self.memory_mcp = MemoryMCPManager()
        self.workflow_executor = WorkflowExecutor()
        self.command_registry = CommandRegistry()
        
    def register_workflow_command(self, workflow_id: str, command: str, config_path: str):
        """Register custom command for workflow"""
        command_entry = {
            "workflow_id": workflow_id,
            "command": command,
            "config_path": config_path,
            "created": datetime.now().isoformat(),
            "status": "registered"
        }
        
        # Store in command registry
        self.command_registry.add_command(command, command_entry)
        
        # Track in Memory MCP
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Command registered: '{command}'"
        )
        
        return command_entry
    
    def execute_workflow_command(self, command: str, args: list = None):
        """Execute workflow by custom command"""
        # Look up workflow by command
        command_entry = self.command_registry.get_command(command)
        if not command_entry:
            raise ValueError(f"Command '{command}' not found")
        
        workflow_id = command_entry["workflow_id"]
        config_path = command_entry["config_path"]
        
        # Load workflow configuration
        with open(config_path) as f:
            config = json.load(f)
        
        # Update status in Memory MCP
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Execution started via command: '{command}'"
        )
        
        # Execute workflow
        return self.workflow_executor.execute_workflow(
            workflow_id=workflow_id,
            config=config,
            args=args or []
        )
    
    def list_available_commands(self):
        """List all registered workflow commands"""
        commands = self.command_registry.list_commands()
        
        # Enhance with status from Memory MCP
        enhanced_commands = []
        for cmd in commands:
            workflow_context = self.memory_mcp.get_workflow_context(
                cmd["workflow_id"]
            )
            cmd["last_status"] = self._extract_last_status(workflow_context)
            enhanced_commands.append(cmd)
        
        return enhanced_commands
    
    def get_workflow_status(self, workflow_id: str):
        """Get detailed workflow status"""
        context = self.memory_mcp.get_workflow_context(workflow_id)
        
        if not context:
            return {"status": "not_found"}
        
        return {
            "workflow_id": workflow_id,
            "status": self._extract_current_status(context),
            "progress": self._calculate_progress(context),
            "last_activity": self._get_last_activity(context),
            "current_phase": self._get_current_phase(context),
            "deliverables": self._list_deliverables(workflow_id)
        }
```

class AgentOrchestrator:
    def __init__(self):
        self.memory_mcp = MemoryMCPManager()
        self.files_api = FilesAPIManager()
        self.tool_manager = ToolManager()
        self.button_manager = ButtonManager()
        
    def execute_workflow_phase(self, workflow_id: str, phase_config: dict):
        """Execute single workflow phase with agent handoff"""
        phase_id = f"phase-{uuid4().hex[:6]}"
        
        # Update workflow state
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Starting phase: {phase_config['name']} ({phase_id})"
        )
        
        # Prepare agent materials
        agent_materials = self._prepare_agent_materials(
            workflow_id=workflow_id,
            phase_id=phase_id,
            phase_config=phase_config
        )
        
        # Create handoff package
        handoff_package = self.files_api.prepare_agent_handoff(
            workflow_id=workflow_id,
            agent_materials=agent_materials
        )
        
        # Execute handoff
        agent_result = self._execute_agent_handoff(
            workflow_id=workflow_id,
            phase_id=phase_id,
            handoff_package=handoff_package,
            phase_config=phase_config
        )
        
        # Process agent return
        return self._process_agent_return(
            workflow_id=workflow_id,
            phase_id=phase_id,
            agent_result=agent_result
        )
    
    def _prepare_agent_materials(self, workflow_id: str, phase_id: str, phase_config: dict):
        """Prepare comprehensive materials for agent"""
        
        # Get workflow context
        workflow_context = self.memory_mcp.get_workflow_context(workflow_id)
        
        # Prepare tool snippets
        required_tools = phase_config.get("tools", [])
        tool_snippets = {}
        
        for tool_name in required_tools:

                tool_name=tool_name,

        return {
            "workflow_id": workflow_id,
            "phase_id": phase_id,
            "briefing": agent_briefing,
            "tools": tool_snippets,
            "context": workflow_context,
            "deliverable_spec": phase_config.get("deliverable", ""),
            "success_criteria": phase_config.get("success_criteria", []),
            "callback_instructions": f"When complete, call back with workflow ID: {workflow_id}"
        }

## Workflow Context
**ID:** {workflow_id}
**Goal:** {workflow_context.get('goal', 'Not specified')}
**Current Phase:** {phase_config['name']}

## Your Mission
{phase_config.get('description', 'Complete the assigned task')}

## Deliverable Required
{phase_config.get('deliverable', 'Produce results as specified')}

## Success Criteria
{chr(10).join(f"- {criteria}" for criteria in phase_config.get('success_criteria', []))}

## Available Tools
{chr(10).join(f"- {tool}" for tool in phase_config.get('tools', []))}

- Save all work using the provided tool snippets
- All files will be automatically saved with workflow tracking
- When complete, call back with workflow ID: {workflow_id}
- Context length limit: {phase_config.get('context_limit', '200k')} tokens

{self._get_previous_phase_summary(workflow_id)}

        self.memory_mcp.update_workflow_state(

            f"Agent handoff executed for phase: {phase_id}"

        # Simulate agent work and return
        return {
            "phase_id": phase_id,
            "status": "completed",
            "execution_results": {
                "files_created": ["example_deliverable.md", "analysis_results.json"],
                "tools_used": phase_config.get("tools", []),
                "execution_time": "45 minutes",
                "quality_score": 0.95
            },
            "callback_data": {
                "workflow_id": workflow_id,
                "agent_notes": "Phase completed successfully with high quality results"

        """Process agent return and determine next steps"""
        
        # Update workflow state with results
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Phase {phase_id} completed: {agent_result['status']}"
        )
        
        # Process any files created
        if agent_result.get("execution_results", {}).get("files_created"):
            for file_name in agent_result["execution_results"]["files_created"]:
                self.memory_mcp.update_workflow_state(
                    workflow_id,
                    f"Deliverable created: {file_name}"
                )
        
        # Determine next phase
        next_phase = self._determine_next_phase(workflow_id, agent_result)
        
        return {
            "phase_completed": phase_id,
            "results": agent_result,
            "next_phase": next_phase,
            "workflow_status": "continuing" if next_phase else "completed"
        }
    
    def handle_agent_callback(self, workflow_id: str, callback_data: dict):
        """Handle agent callback with results"""
        
        # Verify workflow exists
        workflow_context = self.memory_mcp.get_workflow_context(workflow_id)
        if not workflow_context:
            raise ValueError(f"Workflow {workflow_id} not found")
        
        # Process callback
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Agent callback received: {callback_data.get('status', 'unknown')}"
        )
        
        # Continue workflow if needed
        return self._continue_workflow_from_callback(workflow_id, callback_data)

# orchestrator/workflow_state_manager.py
class WorkflowStateManager:
    def __init__(self):
        self.memory_mcp = MemoryMCPManager()
        self.files_api = FilesAPIManager()
        
    def create_workflow_state(self, workflow_id: str, initial_config: dict):
        """Create comprehensive workflow state"""
        state = {
            "workflow_id": workflow_id,
            "created": datetime.now().isoformat(),
            "status": "initialized",
            "current_phase": 0,
            "total_phases": len(initial_config.get("phases", [])),
            "progress_percentage": 0,
            "config": initial_config,
            "execution_log": [],
            "deliverables": [],
            "cost_tracking": {
                "estimated": 0.0,
                "actual": 0.0,
                "breakdown": {}
            }
        }
        
        # Save initial state
        self._save_workflow_state(workflow_id, state)
        
        return state
    
    def update_workflow_progress(self, workflow_id: str, phase_completed: str, results: dict):
        """Update workflow progress and state"""
        state = self._load_workflow_state(workflow_id)
        
        # Update progress
        state["current_phase"] += 1
        state["progress_percentage"] = (state["current_phase"] / state["total_phases"]) * 100
        
        # Add to execution log
        state["execution_log"].append({
            "timestamp": datetime.now().isoformat(),
            "phase": phase_completed,
            "status": results.get("status", "completed"),
            "deliverables": results.get("deliverables", []),
            "cost": results.get("cost", 0.0)
        })
        
        # Update cost tracking
        if results.get("cost"):
            state["cost_tracking"]["actual"] += results["cost"]
            state["cost_tracking"]["breakdown"][phase_completed] = results["cost"]
        
        # Add deliverables
        if results.get("deliverables"):
            state["deliverables"].extend(results["deliverables"])
        
        # Update status
        if state["current_phase"] >= state["total_phases"]:
            state["status"] = "completed"
        else:
            state["status"] = "in_progress"
        
        # Save updated state
        self._save_workflow_state(workflow_id, state)
        
        # Update Memory MCP
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Progress: {state['progress_percentage']:.1f}% - Phase {phase_completed} completed"
        )
        
        return state
    
    def recover_workflow_session(self, workflow_id: str):
        """Recover workflow state after interruption"""
        
        # Get state from Memory MCP
        workflow_context = self.memory_mcp.get_workflow_context(workflow_id)
        if not workflow_context:
            return None
        
        # Load detailed state
        state = self._load_workflow_state(workflow_id)
        
        # Determine recovery point
        recovery_info = {
            "workflow_id": workflow_id,
            "last_completed_phase": state.get("current_phase", 0) - 1,
            "next_phase": state.get("current_phase", 0),
            "progress": state.get("progress_percentage", 0),
            "deliverables_so_far": state.get("deliverables", []),
            "can_resume": state.get("status") in ["in_progress", "paused"],
            "recovery_notes": self._generate_recovery_notes(state)
        }
        
        # Track recovery attempt
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Session recovery attempted - {recovery_info['can_resume']}"
        )
        
        return recovery_info
    
    def _save_workflow_state(self, workflow_id: str, state: dict):
        """Save workflow state to Files API"""
        state_json = json.dumps(state, indent=2)
        
        # Save to Files API (via Code Execution for accessibility)
        file_id = self.files_api.save_workflow_file(
            workflow_id=workflow_id,
            filename="workflow_state.json",
            content=state_json
        )
        
        return file_id
    
    def _load_workflow_state(self, workflow_id: str):
        """Load workflow state from Files API"""
        try:
            state_content = self.files_api.load_workflow_file(
                workflow_id=workflow_id,
                filename="workflow_state.json"
            )
            return json.loads(state_content)
        except FileNotFoundError:
            # Create default state if not found
            return {
                "workflow_id": workflow_id,
                "status": "unknown",
                "current_phase": 0,
                "progress_percentage": 0,
                "execution_log": [],
                "deliverables": []
            }
```
