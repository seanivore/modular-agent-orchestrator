```python
# tools/code_execution/code_execution.py
class CodeExecutionTool:
    def __init__(self):
        self.memory_mcp = MemoryMCPManager()
        self.files_api = FilesAPIManager()
        
    def execute_human_button(self, button_code: str, workflow_id: str, context: dict = None):
        """Execute human button code with workflow tracking"""
        execution_id = f"exec-{uuid4().hex[:8]}"
        
        # Log execution start in Memory MCP
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Code execution started: {execution_id}"
        )
        
        try:
            # Prepare execution environment with workflow context
            exec_env = self._prepare_environment(workflow_id, context)
            
            # Execute code with Claude Code Execution
            result = self._execute_code(button_code, exec_env)
            
            # Save any generated files to Files API (required for later download)
            if result.get('files'):
                file_ids = []
                for file_data in result['files']:
                    file_id = self._save_execution_file(
                        file_data, 
                        workflow_id, 
                        execution_id
                    )
                    file_ids.append(file_id)
                result['file_ids'] = file_ids
            
            # Log successful execution
            self.memory_mcp.update_workflow_state(
                workflow_id,
                f"Code execution completed: {execution_id} -> {len(result.get('file_ids', []))} files"
            )
            
            return {
                "execution_id": execution_id,
                "workflow_id": workflow_id,
                "result": result,
                "status": "success"
            }
            
        except Exception as e:
            # Log execution failure
            self.memory_mcp.update_workflow_state(
                workflow_id,
                f"Code execution failed: {execution_id} -> {str(e)}"
            )
            raise
    
    def _prepare_environment(self, workflow_id: str, context: dict):
        """Prepare execution environment with workflow context"""
        env = {
            'workflow_id': workflow_id,
            'context': context or {},
            'memory_lookup': lambda: self.memory_mcp.get_workflow_context(workflow_id)
        }
        return env
    
    def _save_execution_file(self, file_data: dict, workflow_id: str, execution_id: str):
        """Save execution result file to Files API (required for later access)"""
        filename = f"workflow-{workflow_id}-exec-{execution_id}-{file_data['name']}"
        
        # Upload via Code Execution tool (only way to enable later download)
        file_id = self.files_api.upload_execution_result(
            content=file_data['content'],
            filename=filename,
            workflow_id=workflow_id
        )
        
        return file_id
```

```python
# orchestrator/manager_tools.py (enhanced)
class ToolManager:
    def __init__(self):
        self.discovered_tools = {}
        self.mcp_connector = MCPConnector()
        self.memory_mcp = MemoryMCPManager()
        
    def discover_all_tools(self):
        """Discover tools from multiple sources"""
        tools = {}
        
        # 1. Discover local MAO tools
        local_tools = self._discover_local_tools()
        tools.update(local_tools)
        
        # 2. Discover MCP server tools
        mcp_tools = self.mcp_connector.get_available_tools()
        tools.update(mcp_tools)
        
        # 3. Cache discovery results
        self.discovered_tools = tools
        
        # 4. Log discovery results
        self.memory_mcp.create_entities([{
            "name": "tool-discovery-session",
            "entityType": "system-state",
            "observations": [
                f"Discovered {len(local_tools)} local tools",
                f"Discovered {len(mcp_tools)} MCP tools",
                f"Total tools available: {len(tools)}"
            ]
        }])
        
        return tools
    
    def _discover_local_tools(self):
        """Discover local MAO tools following standardized pattern"""
        tools = {}
        tools_dir = Path("tools")
        
        for tool_dir in tools_dir.iterdir():
            if tool_dir.is_dir() and not tool_dir.name.startswith('_'):
                tool_info = self._analyze_tool_directory(tool_dir)
                if tool_info:
                    tools[tool_dir.name] = tool_info
        
        return tools
    
    def _analyze_tool_directory(self, tool_dir: Path):
        """Analyze tool directory for standardized components"""
        required_files = [
            f"{tool_dir.name}.py",           # Main tool logic
            f"tool_{tool_dir.name}.json",    # Tool configuration
            f"button_{tool_dir.name}.py"     # Button snippet generator
        ]
        
        # Check for standardized structure
        missing_files = []
        for req_file in required_files:
            if not (tool_dir / req_file).exists():
                missing_files.append(req_file)
        
        if missing_files:
            print(f"Tool {tool_dir.name} missing: {missing_files}")
            return None
        
        # Load tool configuration
        config_path = tool_dir / f"tool_{tool_dir.name}.json"
        with open(config_path) as f:
            config = json.load(f)
        
        # Verify button snippet function exists
        button_module = self._load_button_module(tool_dir)
        if not hasattr(button_module, 'create_button_snippet'):
            print(f"Tool {tool_dir.name} missing create_button_snippet function")
            return None
        
        return {
            "type": "local",
            "config": config,
            "button_generator": button_module.create_button_snippet,
            "main_module": tool_dir / f"{tool_dir.name}.py"
        }
    
    def get_tool_for_workflow(self, tool_name: str, workflow_id: str):
        """Get tool configured for specific workflow"""
        if tool_name not in self.discovered_tools:
            # Attempt rediscovery
            self.discover_all_tools()
            
        if tool_name not in self.discovered_tools:
            raise ValueError(f"Tool {tool_name} not available")
        
        tool_info = self.discovered_tools[tool_name]
        
        # Track tool request
        self.memory_mcp.update_workflow_state(
            workflow_id,
            f"Tool requested: {tool_name}"
        )
        
        return tool_info
```

```python
# orchestrator/core.py (enhanced)
class WorkflowOrchestrator:
    def __init__(self):
        self.tool_manager = ToolManager()
        self.button_manager = ButtonManager()
        self.memory_mcp = MemoryMCPManager()
        
        # Discover tools on startup
        self.available_tools = self.tool_manager.discover_all_tools()
    
    def prepare_agent_materials(self, workflow_id: str, phase_config: dict):
        """Prepare materials for agent with dynamic tool access"""
        required_tools = phase_config.get('tools', [])
        agent_materials = {
            "workflow_id": workflow_id,
            "phase": phase_config['name'],
            "tools": {}
        }
        
        # Generate executable button snippets for required tools
        for tool_name in required_tools:
            tool_info = self.tool_manager.get_tool_for_workflow(tool_name, workflow_id)
            
            button_snippet = self.button_manager.create_executable_button_snippet(
                tool_name=tool_name,
                workflow_id=workflow_id,
                agent_context=phase_config.get('context', {})
            )
            
            agent_materials["tools"][tool_name] = {
                "snippet": button_snippet,
                "config": tool_info["config"]
            }
        
        return agent_materials
```
